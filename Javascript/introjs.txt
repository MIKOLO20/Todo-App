  //Data Types
  1. String - "abcde12345", '1234', '', "", "Hi John"
  2. Number - 0123456789
  3. Boolean - true, false
  4. BigInt - 9.99999999999e + 23 (An extremely large number)
  5. Undefined - (It has no value yet)
  6. Null - (purposely declaring a variable empty)
  7. Object - Arrays, Object, Date, Math


  //Arithmetic Operators
  1. + - Addition
  2. - - Subtraction
  3. * - Multiplication
  4. / - Division
  5. % - Modulud (Remainder) 5 % 3 = 2, 4 % 4 = 0


  //Unary Operators 
  1. ++ - Increment by 1 e.g.
  let a = 5
  a++ (a is now 6)
  2. -- - Decrement by 1 e.g.
  lat a = 5
  a -- (a is now 4)

  1. && - add 
  2. || - or
  3. != - not equal to
  4. == checks value alone (not strict) e.g. 5 and "5" is true 
  5. === checks value alone (strict) e.g. 5 and "5" is false
  6. >
  7. <
  8. >=
  9. <=
